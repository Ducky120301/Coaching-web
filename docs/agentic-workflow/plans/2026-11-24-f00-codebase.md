**Goal:** Sửa backend cho khớp BR-f00-codebase-001→020, dựng workspace Angular 22 (admin + client + shared) theo prototype, tự động hóa toàn bộ scenario `features/nen-tang/**`, chạy được bằng docker compose, CI GitHub Actions và Blueprint Render.

```csharp
using System.Net.Http.Headers;
using System.Net.Http.Json;
using Coaching.Application.Auth;
using Coaching.Application.Common;
using Coaching.Infrastructure.Identity;
using Microsoft.AspNetCore.Identity;
using Microsoft.Extensions.DependencyInjection;
using Microsoft.Extensions.Time.Testing;

namespace Coaching.Acceptance.Support;

/// <summary>Trạng thái của 1 scenario (Reqnroll tạo mới cho mỗi scenario).</summary>
public sealed class AcceptanceContext : IDisposable
{
    public const string AdminEmail = "admin@acceptance.local";
    public const string AdminPassword = "Admin@12345";
    public const string AdminName = "Coach Duobexiuuu";

    private ApiHost? _host;

    // Bắt đầu từ giờ thật: trước Task 2 JwtBearer còn so hạn token với đồng hồ hệ thống.
    public FakeTimeProvider Clock { get; } = new(DateTimeOffset.UtcNow);

    public Dictionary<string, string?> Settings { get; } = new()
    {
        ["Jwt:SigningKey"] = "acceptance-signing-key-0123456789abcdef",
        ["AdminSeed:Email"] = AdminEmail,
        ["AdminSeed:Password"] = AdminPassword,
        ["AdminSeed:FullName"] = AdminName,
        ["Email:Provider"] = "Log",
        ["RateLimiting:AuthPermitPerMinute"] = "1000",
    };

    public string ConnectionString { get; } = SqlServerHooks.NewDatabaseConnectionString();

    public Exception? StartupError { get; private set; }

    public HttpResponseMessage? LastResponse { get; set; }

    public AuthTokens? Tokens { get; set; }

    public ApiHost Host
    {
        get
        {
            if (_host is null)
                TryStart();
            return _host ?? throw new InvalidOperationException("API không khởi động được.", StartupError);
        }
    }

    /// <summary>Khởi động (lại) API trên cùng database; lỗi khởi động được ghi vào StartupError.</summary>
    public void TryStart()
    {
        _host?.Dispose();
        _host = null;
        StartupError = null;
        var host = new ApiHost(ConnectionString, Settings, Clock);
        try
        {
            _ = host.Services; // buộc host khởi động → migrate + seed
            _host = host;
        }
        catch (Exception ex)
        {
            StartupError = ex;
            host.Dispose();
        }
    }

    public HttpClient Client() => Host.CreateClient();

    public async Task<HttpResponseMessage> PostJsonAsync(string url, object body, string? ip = null)
    {
        using var request = new HttpRequestMessage(HttpMethod.Post, url) { Content = JsonContent.Create(body) };
        if (ip is not null)
            request.Headers.Add("X-Forwarded-For", ip);
        LastResponse = await Client().SendAsync(request);
        return LastResponse;
    }

    public Task<HttpResponseMessage> LoginAsync(string email, string password, string app, string? ip = null) =>
        PostJsonAsync("/api/auth/login", new { email, password, app }, ip);

    public async Task<AuthTokens> LoginOkAsync(string email, string password, string app)
    {
        var response = await LoginAsync(email, password, app);
        response.EnsureSuccessStatusCode();
        Tokens = (await response.Content.ReadFromJsonAsync<AuthTokens>())!;
        return Tokens;
    }

    public async Task<HttpResponseMessage> GetAsync(string url, string? accessToken)
    {
        using var request = new HttpRequestMessage(HttpMethod.Get, url);
        if (accessToken is not null)
            request.Headers.Authorization = new AuthenticationHeaderValue("Bearer", accessToken);
        LastResponse = await Client().SendAsync(request);
        return LastResponse;
    }

    /// <summary>Tạo tài khoản học viên trực tiếp qua Identity (luồng mời làm ở F01).</summary>
    public async Task<(string Email, string Password)> CreateStudentAsync()
    {
        var email = $"hocvien-{Guid.NewGuid():N}@acceptance.local";
        const string password = "Student@123";
        await using var scope = Host.Services.CreateAsyncScope();
        var users = scope.ServiceProvider.GetRequiredService<UserManager<AppUser>>();
        var user = new AppUser { UserName = email, Email = email, FullName = "Học viên Acceptance", EmailConfirmed = true };
        var created = await users.CreateAsync(user, password);
        if (!created.Succeeded)
            throw new InvalidOperationException(string.Join("; ", created.Errors.Select(e => e.Description)));
        await users.AddToRoleAsync(user, Roles.Student);
        return (email, password);
    }

    public void Dispose() => _host?.Dispose();
}

namespace Coaching.Acceptance.Steps;

[Binding]
public sealed class AccessSteps(AcceptanceContext ctx)

        Assert.Equal(AcceptanceContext.AdminName, profile.DisplayName);

using System.IdentityModel.Tokens.Jwt;
using System.Text;
using Coaching.Application.Common;
using Coaching.Infrastructure.Identity;
using Microsoft.AspNetCore.Authentication.JwtBearer;
using Microsoft.Extensions.Options;
using Microsoft.IdentityModel.Tokens;

namespace Coaching.Api.Auth;

internal static class AuthenticationSetup
{
    private static readonly TimeSpan ClockSkew = TimeSpan.FromSeconds(30);

    public static IServiceCollection AddJwtAuth(this IServiceCollection services, IConfiguration configuration)
    {
        var jwt = configuration.GetSection(JwtOptions.SectionName).Get<JwtOptions>() ?? new JwtOptions();

        services
            .AddAuthentication(JwtBearerDefaults.AuthenticationScheme)
            .AddJwtBearer(options =>
            {
                options.MapInboundClaims = false;
                options.TokenValidationParameters = new TokenValidationParameters
                {
                    ValidateIssuer = true,
                    ValidIssuer = jwt.Issuer,
                    ValidateAudience = true,
                    ValidAudience = jwt.Audience,
                    ValidateIssuerSigningKey = true,
                    IssuerSigningKey = new SymmetricSecurityKey(Encoding.UTF8.GetBytes(jwt.SigningKey)),
                    ValidateLifetime = true,
                    ClockSkew = ClockSkew,
                    NameClaimType = JwtRegisteredClaimNames.Sub,
                    RoleClaimType = JwtTokenGenerator.RoleClaimType,
                };
            });

        // @ADR-f00-codebase-005 @BR-f00-codebase-003: hạn token so với TimeProvider của DI (test tua được giờ)
        services.AddOptions<JwtBearerOptions>(JwtBearerDefaults.AuthenticationScheme)
            .Configure<TimeProvider>((options, time) =>
                options.TokenValidationParameters.LifetimeValidator = (notBefore, expires, _, _) =>
                {
                    var now = time.GetUtcNow().UtcDateTime;
                    return (notBefore is null || notBefore.Value <= now + ClockSkew)
                        && expires is not null && expires.Value > now - ClockSkew;
                });

        services.AddAuthorizationBuilder()
            .AddPolicy(Policies.AdminOnly, p => p.RequireAuthenticatedUser().RequireRole(Roles.Admin))
            .AddPolicy(Policies.StudentOnly, p => p.RequireAuthenticatedUser().RequireRole(Roles.Student));

        return services;
    }
}

using DotNet.Testcontainers.Builders;
using DotNet.Testcontainers.Networks;
using Microsoft.Data.SqlClient;
using Testcontainers.MsSql;

namespace Coaching.TestSupport;

/// <summary>
/// @ADR-f00-codebase-011: SQL Server Testcontainers + Liquibase chạy 1 lần vào database mẫu rồi backup;
/// mỗi test/scenario restore thành 1 database mới có sẵn schema.
/// </summary>
public sealed class DatabaseFixture : IAsyncDisposable
{
    private const string TemplateDb = "coaching_template";
    private const string BackupFile = "/var/opt/mssql/data/coaching_template.bak";
    private const string LiquibaseImage = "liquibase/liquibase:4.33.0";

    private readonly INetwork _network = new NetworkBuilder().Build();
    private readonly MsSqlContainer _sql;

    public DatabaseFixture()
    {
        _sql = new MsSqlBuilder("mcr.microsoft.com/mssql/server:2022-latest")
            .WithNetwork(_network)
            .WithNetworkAliases("sql")
            .Build();
    }

    public async Task StartAsync()
    {
        await _network.CreateAsync();
        await _sql.StartAsync();
        await ExecuteAsync("master", $"CREATE DATABASE [{TemplateDb}]");
        await RunLiquibaseUpdateAsync();
        await ExecuteAsync("master", $"BACKUP DATABASE [{TemplateDb}] TO DISK = N'{BackupFile}' WITH INIT");
    }

    public async Task<string> NewDatabaseAsync()
    {
        var name = $"t_{Guid.NewGuid():N}";
        await ExecuteAsync("master",
            $"RESTORE DATABASE [{name}] FROM DISK = N'{BackupFile}' WITH " +
            $"MOVE N'{TemplateDb}' TO N'/var/opt/mssql/data/{name}.mdf', " +
            $"MOVE N'{TemplateDb}_log' TO N'/var/opt/mssql/data/{name}_log.ldf'");
        return ConnectionString(name);
    }

    public async ValueTask DisposeAsync()
    {
        await _sql.DisposeAsync();
        await _network.DisposeAsync();
    }

    private string ConnectionString(string database) =>
        new SqlConnectionStringBuilder(_sql.GetConnectionString()) { InitialCatalog = database }.ConnectionString;

    private async Task ExecuteAsync(string database, string sql)
    {
        await using var connection = new SqlConnection(ConnectionString(database));
        await connection.OpenAsync();
        await using var command = new SqlCommand(sql, connection) { CommandTimeout = 120 };
        await command.ExecuteNonQueryAsync();
    }

    private async Task RunLiquibaseUpdateAsync()
    {
        var liquibase = new ContainerBuilder()
            .WithImage(LiquibaseImage)
            .WithNetwork(_network)
            .WithBindMount(FindChangelogDirectory(), "/liquibase/changelog")
            .WithCommand(
                "--search-path=/liquibase/changelog",
                "--changelog-file=db.changelog-master.xml",
                $"--url=jdbc:sqlserver://sql:1433;databaseName={TemplateDb};encrypt=false;trustServerCertificate=true",
                "--username=sa",
                $"--password={MsSqlBuilder.DefaultPassword}",
                "update")
            .Build();
        await using (liquibase)
        {
            await liquibase.StartAsync();
            var exitCode = await liquibase.GetExitCodeAsync();
            if (exitCode != 0)
            {
                var (stdout, stderr) = await liquibase.GetLogsAsync();
                throw new InvalidOperationException($"Liquibase update lỗi (exit {exitCode}):\n{stdout}\n{stderr}");
            }
        }
    }

    private static string FindChangelogDirectory()
    {
        for (var dir = new DirectoryInfo(AppContext.BaseDirectory); dir is not null; dir = dir.Parent)
        {
            var candidate = Path.Combine(dir.FullName, "database", "changelog");
            if (Directory.Exists(candidate))
                return candidate;
        }
        throw new DirectoryNotFoundException("Không tìm thấy thư mục database/changelog từ thư mục test.");
    }
}

using System.Globalization;
using System.Threading.RateLimiting;
using Microsoft.AspNetCore.Mvc;
using Microsoft.AspNetCore.RateLimiting;

namespace Coaching.Api.Common;

internal static class RateLimitPolicies
{
    public const string Auth = "auth";

    /// <summary>@BR-f00-codebase-008: mỗi IP thật tối đa N yêu cầu đăng nhập/gia hạn mỗi phút (mặc định 10).</summary>
    public static IServiceCollection AddCoachingRateLimiting(this IServiceCollection services, IConfiguration configuration)
    {
        var permitPerMinute = configuration.GetValue("RateLimiting:AuthPermitPerMinute", 10);
        services.AddRateLimiter(options =>
        {
            options.RejectionStatusCode = StatusCodes.Status429TooManyRequests;
            options.AddPolicy(Auth, httpContext => RateLimitPartition.GetFixedWindowLimiter(
                httpContext.Connection.RemoteIpAddress?.ToString() ?? "unknown",
                _ => new FixedWindowRateLimiterOptions { PermitLimit = permitPerMinute, Window = TimeSpan.FromMinutes(1) }));
            options.OnRejected = async (context, cancellationToken) =>
            {
                var seconds = context.Lease.TryGetMetadata(MetadataName.RetryAfter, out var retryAfter)
                    ? Math.Max(1, (int)Math.Ceiling(retryAfter.TotalSeconds))
                    : 60;
                var response = context.HttpContext.Response;
                response.StatusCode = StatusCodes.Status429TooManyRequests;
                response.Headers.RetryAfter = seconds.ToString(CultureInfo.InvariantCulture);
                var problem = new ProblemDetails
                {
                    Status = StatusCodes.Status429TooManyRequests,
                    Title = $"Bạn thao tác quá nhanh. Vui lòng thử lại sau {seconds} giây.",
                    Type = "rate_limited",
                };
                problem.Extensions["code"] = "rate_limited";
                await response.WriteAsJsonAsync(problem, cancellationToken);
            };
        });
        return services;
    }
}

using Coaching.Application.Abstractions;
using Coaching.Application.Common;
using Coaching.Domain.Coaches;
using Coaching.Infrastructure.Identity;
using Coaching.Infrastructure.Persistence;
using Microsoft.AspNetCore.Identity;
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.DependencyInjection;
using Microsoft.Extensions.Logging;
using Microsoft.Extensions.Options;
using DomainEmail = Coaching.Domain.Common.Email;

namespace Coaching.Infrastructure.Seeding;

/// <summary>Kiểm schema (do Liquibase tạo), tạo role, admin + hồ sơ coach và (tuỳ môi trường) học viên demo. Idempotent.</summary>
public static class DatabaseInitializer
{
    public const int MinPasswordLength = 8;

    public static async Task InitializeAsync(IServiceProvider services, CancellationToken cancellationToken = default)
    {
        await using var scope = services.CreateAsyncScope();
        var sp = scope.ServiceProvider;
        var logger = sp.GetRequiredService<ILoggerFactory>().CreateLogger(typeof(DatabaseInitializer));
        var db = sp.GetRequiredService<AppDbContext>();

        var adminSeed = sp.GetRequiredService<IOptions<AdminSeedOptions>>().Value;
        ValidateSeedCredentials("AdminSeed", adminSeed.Email, adminSeed.Password);

        // @ADR-f00-codebase-011: schema do Liquibase quản lý; API không migrate, chỉ kiểm schema đã có rồi seed
        await EnsureSchemaExistsAsync(db, cancellationToken);

        var roleManager = sp.GetRequiredService<RoleManager<IdentityRole<Guid>>>();
        foreach (var role in Roles.All)
        {
            if (!await roleManager.RoleExistsAsync(role))
                EnsureSucceeded(await roleManager.CreateAsync(new IdentityRole<Guid>(role)), $"tạo role {role}");
        }

        var userManager = sp.GetRequiredService<UserManager<AppUser>>();
        var clock = sp.GetRequiredService<IClock>();

        // @BR-f00-codebase-016 @BR-f00-codebase-019: chỉ tạo khi chưa có; không đổi mật khẩu admin đã tồn tại
        var adminEmail = DomainEmail.Create(adminSeed.Email);
        var admin = await EnsureUserAsync(userManager, clock, adminEmail.Value, adminSeed.Password!, adminSeed.FullName, Roles.Admin, logger);
        if (!await db.Coaches.AnyAsync(c => c.UserId == admin.Id, cancellationToken))
        {
            db.Coaches.Add(Coach.Create(admin.Id, adminSeed.FullName, adminEmail));
            await db.SaveChangesAsync(cancellationToken);
        }

        // @BR-f00-codebase-020
        var demoSeed = sp.GetRequiredService<IOptions<DemoSeedOptions>>().Value;
        if (demoSeed.Enabled)
        {
            ValidateSeedCredentials("DemoSeed", demoSeed.Email, demoSeed.Password);
            var demoEmail = DomainEmail.Create(demoSeed.Email);
            await EnsureUserAsync(userManager, clock, demoEmail.Value, demoSeed.Password!, demoSeed.FullName, Roles.Student, logger);
        }
    }

    private static async Task EnsureSchemaExistsAsync(AppDbContext db, CancellationToken cancellationToken)
    {
        var applied = await db.Database
            .SqlQueryRaw<int>("SELECT COUNT(*) AS [Value] FROM sys.tables WHERE name = 'DATABASECHANGELOG'")
            .SingleAsync(cancellationToken);
        if (applied == 0)
            throw new InvalidOperationException(
                "Database chưa có schema. Chạy Liquibase (`liquibase update`, xem database/) trước khi khởi động API.");
    }

    // @BR-f00-codebase-017 @BR-f00-codebase-018: thiếu cấu hình hoặc mật khẩu < 8 ký tự → không khởi động
    private static void ValidateSeedCredentials(string section, string? email, string? password)
    {
        if (string.IsNullOrWhiteSpace(email))
            throw new InvalidOperationException($"Thiếu cấu hình {section}:Email — hệ thống không khởi động.");
        if (string.IsNullOrEmpty(password))
            throw new InvalidOperationException($"Thiếu cấu hình {section}:Password — hệ thống không khởi động.");
        if (password.Length < MinPasswordLength)
            throw new InvalidOperationException(
                $"{section}:Password phải có ít nhất {MinPasswordLength} ký tự — hệ thống không khởi động.");
    }

    private static async Task<AppUser> EnsureUserAsync(UserManager<AppUser> userManager, IClock clock,
        string email, string password, string fullName, string role, ILogger logger)
    {
        var user = await userManager.FindByEmailAsync(email);
        if (user is null)
        {
            user = new AppUser
            {
                UserName = email,
                Email = email,
                EmailConfirmed = true,
                FullName = fullName,
                CreatedAt = clock.UtcNow,
            };
            EnsureSucceeded(await userManager.CreateAsync(user, password), $"tạo tài khoản {email}");
            logger.LogInformation("Đã tạo tài khoản {Email} (role {Role}).", email, role);
        }

        if (!await userManager.IsInRoleAsync(user, role))
            EnsureSucceeded(await userManager.AddToRoleAsync(user, role), $"gán role {role} cho {email}");
        return user;
    }

    private static void EnsureSucceeded(IdentityResult result, string action)
    {
        if (!result.Succeeded)
            throw new InvalidOperationException(
                $"Không thể {action}: {string.Join("; ", result.Errors.Select(e => e.Description))}");
    }
}

import { Injectable, OnDestroy, inject } from '@angular/core';
import { CLIENT_APP } from './client-app';

// @ADR-f00-codebase-002 @BR-f00-codebase-010 @BR-f00-codebase-014
@Injectable({ providedIn: 'root' })
export class RefreshCoordinator implements OnDestroy {
  private readonly app = inject(CLIENT_APP);
  private readonly channel: BroadcastChannel | null =
    typeof BroadcastChannel === 'undefined' ? null : new BroadcastChannel(`dbx-auth-${this.app}`);
  private inFlight: Promise<string> | null = null;

  /** Single-flight trong tab + khóa giữa các tab (Web Locks). Task phải đọc refresh token mới nhất bên trong. */
  run(task: () => Promise<string>): Promise<string> {
    this.inFlight ??= this.withCrossTabLock(task).finally(() => (this.inFlight = null));
    return this.inFlight;
  }

  broadcastLogout(): void {
    this.channel?.postMessage('logout');
  }

  onLoggedOutElsewhere(handler: () => void): void {
    this.channel?.addEventListener('message', (event) => {
      if (event.data === 'logout') handler();
    });
  }

  ngOnDestroy(): void {
    this.channel?.close();
  }

  private withCrossTabLock(task: () => Promise<string>): Promise<string> {
    const locks = typeof navigator === 'undefined' ? undefined : navigator.locks;
    return locks ? locks.request(`dbx-refresh-${this.app}`, () => task()) : task();
  }
}

import { HttpClient, HttpContext } from '@angular/common/http';
import { Injectable, computed, inject, signal } from '@angular/core';
import { Router } from '@angular/router';
import { firstValueFrom } from 'rxjs';
import { APP_CONFIG } from '../config/app-config';
import { CLIENT_APP } from './client-app';
import { SKIP_AUTH } from './skip-auth';
import { AuthTokens, AuthUser } from './models';
import { RefreshCoordinator } from './refresh-coordinator';
import { TokenStore } from './token-store';

@Injectable({ providedIn: 'root' })
export class AuthService {
  private readonly http = inject(HttpClient);
  private readonly config = inject(APP_CONFIG);
  private readonly app = inject(CLIENT_APP);
  private readonly store = inject(TokenStore);
  private readonly coordinator = inject(RefreshCoordinator);
  private readonly router = inject(Router);
  private readonly noAuth = { context: new HttpContext().set(SKIP_AUTH, true) };

  private readonly currentUser = signal<AuthUser | null>(null);
  readonly user = this.currentUser.asReadonly();
  readonly isAuthenticated = computed(() => this.currentUser() !== null);

  constructor() {
    // @BR-f00-codebase-014: tab khác đăng xuất → tab này cũng kết thúc phiên
    this.coordinator.onLoggedOutElsewhere(() => this.endSession(true));
  }

  apiUrl(path: string): string {
    return `${this.config.apiBaseUrl}${path}`;
  }

  accessToken(): string | null {
    return this.store.accessToken();
  }

  async login(email: string, password: string): Promise<void> {
    const tokens = await firstValueFrom(
      this.http.post<AuthTokens>(this.apiUrl('/api/auth/login'), { email, password, app: this.app }, this.noAuth),
    );
    this.accept(tokens);
  }

  /** Khôi phục phiên khi mở lại web (@BR-f00-codebase-010). */
  async restore(): Promise<void> {
    if (!this.store.refreshToken()) return;
    try {
      await this.refresh();
    } catch {
      this.endSession(false);
    }
  }

  /** @BR-f00-codebase-010: gia hạn, chỉ 1 lần dù nhiều request/tab cùng cần. */
  refresh(): Promise<string> {
    return this.coordinator.run(async () => {
      const refreshToken = this.store.refreshToken(); // đọc lại sau khi có khóa: tab khác có thể đã đổi
      if (!refreshToken) throw new Error('Không có phiên để gia hạn');
      const tokens = await firstValueFrom(
        this.http.post<AuthTokens>(this.apiUrl('/api/auth/refresh'), { refreshToken }, this.noAuth),
      );
      this.accept(tokens);
      return tokens.accessToken;
    });
  }

  async logout(): Promise<void> {
    const refreshToken = this.store.refreshToken();
    this.endSession(false);
    this.coordinator.broadcastLogout();
    if (refreshToken) {
      try {
        await firstValueFrom(this.http.post(this.apiUrl('/api/auth/logout'), { refreshToken }, this.noAuth));
      } catch {
        /* phiên phía máy đã xoá; server tự hết hạn */
      }
    }
    await this.router.navigateByUrl('/login');
  }

  handleSessionExpired(): void {
    this.endSession(true);
  }

  private accept(tokens: AuthTokens): void {
    this.store.save(tokens);
    this.currentUser.set(tokens.user);
  }

  private endSession(navigate: boolean): void {
    this.store.clear();
    this.currentUser.set(null);
    if (navigate) void this.router.navigateByUrl('/login');
  }
}

import { HttpErrorResponse, HttpInterceptorFn, HttpRequest } from '@angular/common/http';
import { inject } from '@angular/core';
import { catchError, from, switchMap, throwError } from 'rxjs';
import { APP_CONFIG } from '../config/app-config';
import { AuthService } from './auth.service';
import { SKIP_AUTH } from './skip-auth';

function withBearer(req: HttpRequest<unknown>, token: string | null): HttpRequest<unknown> {
  return token ? req.clone({ setHeaders: { Authorization: `Bearer ${token}` } }) : req;
}

// @BR-f00-codebase-010: 401 → gia hạn 1 lần → gửi lại request gốc
export const authInterceptor: HttpInterceptorFn = (req, next) => {
  const auth = inject(AuthService);
  const { apiBaseUrl } = inject(APP_CONFIG);
  if (req.context.get(SKIP_AUTH) || !req.url.startsWith(`${apiBaseUrl}/api/`)) {
    return next(req);
  }
  return next(withBearer(req, auth.accessToken())).pipe(
    catchError((error: unknown) => {
      if (!(error instanceof HttpErrorResponse) || error.status !== 401) {
        return throwError(() => error);
      }
      return from(auth.refresh()).pipe(
        catchError((refreshError: unknown) => {
          auth.handleSessionExpired();
          return throwError(() => refreshError);
        }),
        switchMap((token) => next(withBearer(req, token))),
      );
    }),
  );
};

import { HttpErrorResponse } from '@angular/common/http';
import { ChangeDetectionStrategy, Component, inject, signal } from '@angular/core';
import { FormsModule } from '@angular/forms';
import { ActivatedRoute, Router } from '@angular/router';
import { AuthService } from '../auth/auth.service';
import { safeReturnUrl } from '../auth/return-url';

function messageFor(error: unknown): string {
  if (error instanceof HttpErrorResponse) {
    if (error.status === 0) return 'Không kết nối được hệ thống. Vui lòng thử lại.';
    const title = (error.error as { title?: unknown } | null)?.title;
    if ([400, 401, 403, 429].includes(error.status) && typeof title === 'string') return title;
  }
  return 'Đã có lỗi xảy ra. Vui lòng thử lại sau.';
}

@Component({
  selector: 'dbx-login-page',
  imports: [FormsModule],
  templateUrl: './login-page.component.html',
  styleUrl: './login-page.component.scss',
  changeDetection: ChangeDetectionStrategy.OnPush,
})
export class LoginPageComponent {
  private readonly auth = inject(AuthService);
  private readonly router = inject(Router);
  private readonly route = inject(ActivatedRoute);

  email = '';
  password = '';
  readonly submitting = signal(false);
  readonly error = signal<string | null>(null);

  async submit(): Promise<void> {
    if (this.submitting()) return;
    this.submitting.set(true);
    this.error.set(null);
    try {
      await this.auth.login(this.email, this.password);
      await this.router.navigateByUrl(safeReturnUrl(this.route.snapshot.queryParamMap.get('returnUrl')));
    } catch (e) {
      this.error.set(messageFor(e));
    } finally {
      this.submitting.set(false);
    }
  }
}

// @ADR-f00-codebase-009: chụp cùng vùng ở prototype và app, tô kín mask trên cả hai, so bằng pixelmatch
import { Page } from 'playwright';
import pixelmatch from 'pixelmatch';
import { PNG } from 'pngjs';

export interface Rect { x: number; y: number; width: number; height: number; }

export async function settle(page: Page): Promise<void> {
  await page.waitForLoadState('networkidle');
  await page.evaluate(() => document.fonts.ready);
  await page.addStyleTag({ content: '#proto-ui{display:none!important} *{caret-color:transparent!important}' });
}

export async function captureRegion(page: Page, regionSelector: string, maskSelectors: string[]) {
  await settle(page);
  const region = page.locator(regionSelector).first();
  const box = (await region.boundingBox())!;
  const masks: Rect[] = [];
  for (const sel of maskSelectors) {
    for (const el of await page.locator(`${regionSelector} ${sel}`).all()) {
      const b = await el.boundingBox();
      if (b) masks.push({ x: b.x - box.x, y: b.y - box.y, width: b.width, height: b.height });
    }
  }
  const png = PNG.sync.read(await region.screenshot({ animations: 'disabled' }));
  return { png, masks };
}

function padTo(src: PNG, width: number, height: number): PNG {
  const out = new PNG({ width, height });
  out.data.fill(255);
  PNG.bitblt(src, out, 0, 0, src.width, src.height, 0, 0);
  return out;
}

function fill(img: PNG, rect: Rect): void {
  const x0 = Math.max(0, Math.floor(rect.x)), y0 = Math.max(0, Math.floor(rect.y));
  const x1 = Math.min(img.width, Math.ceil(rect.x + rect.width)), y1 = Math.min(img.height, Math.ceil(rect.y + rect.height));
  for (let y = y0; y < y1; y++) for (let x = x0; x < x1; x++) {
    const i = (img.width * y + x) << 2;
    img.data[i] = 255; img.data[i + 1] = 0; img.data[i + 2] = 255; img.data[i + 3] = 255;
  }
}

export function diffRatio(a: PNG, b: PNG, masks: Rect[]): { ratio: number; diff: PNG } {
  const width = Math.max(a.width, b.width), height = Math.max(a.height, b.height);
  const pa = padTo(a, width, height), pb = padTo(b, width, height);
  for (const m of masks) { fill(pa, m); fill(pb, m); }
  const diff = new PNG({ width, height });
  const changed = pixelmatch(pa.data, pb.data, diff.data, width, height, { threshold: 0.1 });
  return { ratio: changed / (width * height), diff };
}
```
